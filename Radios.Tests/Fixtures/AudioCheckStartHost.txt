    public static async Task<int[]> Run(string scenario)
    {
        RecorderControls.Spoken = 0;
        var h = new Host();
        // A tune that comes up during the count (#860): [keys while counting, keys, record arms, said].
        if (scenario == "tune-during-count")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.RecordPlayback, AudioCheckPowerModes.LowPower, 1);
            var pending = h.StartAudioCheckSessionAsync(s);
            await Task.Delay(20);
            int whileCounting = h.ptt.Keys;
            h._rig.Tuning = true;
            h.Countdown.SetResult(true);
            await pending;
            return new[] { whileCounting, h.ptt.Keys, h._rig.RecordStarts, RecorderControls.Spoken };
        }
        // A power the radio has not reported (#859): [keys, power writes, restores registered, said].
        if (scenario == "power-unreported")
        {
            h._rig.CheckPowersReported = false;
            h._rig.RadioModel = "FLEX-6300";
            h._rig.XmitPower = 0;
            int writes = 0;
            h._rig.PowerChanged = () => writes++;
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.Monitor, AudioCheckPowerModes.LowPower, 10);
            await h.StartAudioCheckSessionAsync(s);
            return new[] { h.ptt.Keys, writes, h.Restores.Count, RecorderControls.Spoken };
        }
        // The power floor (#859), scenario "power:<model>:<method>:<mode>:<power>:<dummy load>":
        // [power at the key, dummy load at the key (0 or 1), keys, power after the restores].
        if (scenario.StartsWith("power:"))
        {
            string[] p = scenario.Split(':');
            h._rig.RadioModel = p[1];
            h._rig.XmitPower = int.Parse(p[4]);
            if (p[5] == "1") h._rig.DummyLoadMode = true;
            int atKey = -1, dummyAtKey = -1;
            h.ptt.Keyed = () => { atKey = h._rig.XmitPower; dummyAtKey = h._rig.DummyLoadMode ? 1 : 0; };
            var s = new AudioCheckSession(h, h._rig, h.ptt, Enum.Parse<AudioCheckListenMethods>(p[2]),
                Enum.Parse<AudioCheckPowerModes>(p[3]), 10);
            var pending = h.StartAudioCheckSessionAsync(s);
            if (p[2] == nameof(AudioCheckListenMethods.RecordPlayback)) h._rig.Record.SetResult(new(true, "recorded", "slice"));
            await pending;
            // The operator releases their own dummy load before the check ends.
            if (p.Length > 6) { h._rig.DummyLoadMode = false; h._rig.XmitPower = 42; }
            foreach (var restore in h.Restores) restore();
            return new[] { atKey, dummyAtKey, h.ptt.Keys, h._rig.XmitPower };
        }
        // The countdown seam (#835, Track C): [keys while the count still
        // plays, keys at the end, 1 if a count was begun without being ended,
        // stale recorder speech].
        // Repair round 1: the place changed during the count (#863 B2), or the
        // link dropped during it (#596 B2): [keys while counting, keys, counts, said].
        if (scenario is "place-changed-during-count" or "link-dropped-during-count")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.Monitor, AudioCheckPowerModes.LowPower, 1);
            var pending = h.StartAudioCheckSessionAsync(s);
            await Task.Delay(20);
            int whileCounting = h.ptt.Keys;
            if (scenario == "place-changed-during-count") h.PlaceStands = false;
            else h._rig.IsConnected = false;
            h.Countdown.SetResult(true);
            await pending;
            return new[] { whileCounting, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario is "countdown-holds-key" or "countdown-cancelled")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.Monitor, AudioCheckPowerModes.LowPower, 1);
            var pending = h.StartAudioCheckSessionAsync(s);
            await Task.Delay(20);
            int whileCounting = h.ptt.Keys;
            if (scenario == "countdown-cancelled") h.CancelCheckStart(); else h.Countdown.SetResult(true);
            await pending;
            return new[] { whileCounting, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario == "countdown-before-setup")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            h.Countdown.SetResult(true);
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.RecordPlayback, AudioCheckPowerModes.LowPower, 1);
            var pending = h.StartAudioCheckSessionAsync(s);
            await Task.Delay(20);
            int whileSettingUp = h.ptt.Keys;
            h._rig.Record.SetResult(new(true, "recorded", "slice"));
            await pending;
            return new[] { whileSettingUp, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario == "countdown-records-after-count")
        {
            // [recorder arms asked for while the count plays, keys at the end, ...]
            h.Countdown = new TaskCompletionSource<bool>();
            var s = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.RecordPlayback, AudioCheckPowerModes.LowPower, 1);
            var pending = h.StartAudioCheckSessionAsync(s);
            await Task.Delay(20);
            int armedWhileCounting = h._rig.RecordStarts;
            h.Countdown.SetResult(true);
            h._rig.Record.SetResult(new(true, "recorded", "slice"));
            await pending;
            return new[] { armedWhileCounting, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario == "countdown-loopback")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            var pending = h.StartLoopbackCheckAsync();
            h._rig.Arrange.SetResult(true);
            h._rig.Record.SetResult(new(true, "recorded", "slice"));
            await Task.Delay(20);
            int whileCounting = h.ptt.Keys;
            h.Countdown.SetResult(true);
            await pending;
            return new[] { whileCounting, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario == "countdown-off-air")
        {
            h.Countdown = new TaskCompletionSource<bool>();
            var pending = h.StartOffAirAsync(h.ptt);
            h._rig.OffAir.SetResult(new FakeRun());
            await Task.Delay(20);
            int whileCounting = h.ptt.Keys;
            h.Countdown.SetResult(true);
            await pending;
            return new[] { whileCounting, h.ptt.Keys, h.CountdownsBegun == h.CountdownsEnded ? 0 : 1, RecorderControls.Spoken };
        }
        if (scenario is "cancel-off-air" or "off-air-excludes-ordinary")
        {
            var pending = h.StartOffAirAsync(h.ptt);
            var other = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.Monitor, AudioCheckPowerModes.LowPower, 1);
            await h.StartAudioCheckSessionAsync(other);
            if (scenario == "cancel-off-air") h.CancelCheckStart();
            var run = new FakeRun();
            h._rig.OffAir.SetResult(run);
            await pending;
            return new[] { h.ptt.Keys, h.ptt.Keys, scenario == "cancel-off-air" && !run.Stopped ? 1 : 0, RecorderControls.Spoken };
        }
        if (scenario == "cancel-loopback")
        {
            var pending = h.StartLoopbackCheckAsync();
            h.CancelCheckStart();
            h._rig.Arrange.SetResult(true);
            await pending;
            return new[] { 0, h.ptt.Keys, h._rig.LoopbackArranged ? 1 : 0, RecorderControls.Spoken };
        }
        if (scenario == "cancel-before-key") h._rig.PowerChanged = h.CancelCheckStart;
        var first = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.RecordPlayback, AudioCheckPowerModes.LowPower, 1);
        var ordinary = h.StartAudioCheckSessionAsync(first);
        var loopback = scenario == "cross-door" ? h.StartLoopbackCheckAsync() : Task.CompletedTask;
        if (scenario is "cancel-record" or "cancel-record-failed") h.CancelCheckStart();
        h._rig.Record.SetResult(new(scenario != "cancel-record-failed", "recorded", "slice"));
        await ordinary;
        int before = h.ptt.Keys;
        h._session?.StopCheck();
        h._rig.Arrange.SetResult(true);
        await loopback;
        return new[] { before, h.ptt.Keys, h._rig.LoopbackArranged ? 1 : 0, RecorderControls.Spoken };
    }
