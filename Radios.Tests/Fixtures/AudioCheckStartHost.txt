    public static async Task<int[]> Run(string scenario)
    {
        RecorderControls.Spoken = 0;
        var h = new Host();
        if (scenario is "cancel-off-air" or "off-air-excludes-ordinary")
        {
            var pending = h.StartOffAirAsync(h.ptt);
            var other = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.Monitor, AudioCheckPowerModes.LowPower, 1);
            await h.StartAudioCheckSessionAsync(other);
            if (scenario == "cancel-off-air") h.CancelCheckStart();
            var run = new FakeRun();
            h._rig.OffAir.SetResult(run);
            await pending;
            return new[] { h.ptt.Keys, h.ptt.Keys, scenario == "cancel-off-air" && !run.Stopped ? 1 : 0, RecorderControls.Spoken };
        }
        if (scenario == "cancel-loopback")
        {
            var pending = h.StartLoopbackCheckAsync();
            h.CancelCheckStart();
            h._rig.Arrange.SetResult(true);
            await pending;
            return new[] { 0, h.ptt.Keys, h._rig.LoopbackArranged ? 1 : 0, RecorderControls.Spoken };
        }
        if (scenario == "cancel-before-key") h._rig.PowerChanged = h.CancelCheckStart;
        var first = new AudioCheckSession(h, h._rig, h.ptt, AudioCheckListenMethods.RecordPlayback, AudioCheckPowerModes.LowPower, 1);
        var ordinary = h.StartAudioCheckSessionAsync(first);
        var loopback = scenario == "cross-door" ? h.StartLoopbackCheckAsync() : Task.CompletedTask;
        if (scenario is "cancel-record" or "cancel-record-failed") h.CancelCheckStart();
        h._rig.Record.SetResult(new(scenario != "cancel-record-failed", "recorded", "slice"));
        await ordinary;
        int before = h.ptt.Keys;
        h._session?.StopCheck();
        h._rig.Arrange.SetResult(true);
        await loopback;
        return new[] { before, h.ptt.Keys, h._rig.LoopbackArranged ? 1 : 0, RecorderControls.Spoken };
    }
